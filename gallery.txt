<script>
  (function () {
    function initGalleryLightbox() {
      var root = document.querySelector("#catalog-page");
      if (!root) return;

      var items = root.querySelectorAll(".gallery-item img");
      var lightbox = root.querySelector("#gallery-lightbox");
      var lightboxImg = root.querySelector("#gallery-lightbox-img");
      var backdrop = root.querySelector(".glb-backdrop");
      var btnClose = root.querySelector(".glb-close");

      if (!lightbox || !lightboxImg) return;

      function openLightbox(src) {
        lightboxImg.src = src;   // pakai link yang sama dengan thumbnail
        lightbox.classList.add("is-open");
      }

      function closeLightbox() {
        lightbox.classList.remove("is-open");
        lightboxImg.src = "";
      }

      items.forEach(function (img) {
        img.addEventListener("click", function () {
          openLightbox(img.src);
        });
      });

      backdrop.addEventListener("click", closeLightbox);
      btnClose.addEventListener("click", closeLightbox);

      document.addEventListener("keydown", function (e) {
        if (e.key === "Escape" && lightbox.classList.contains("is-open")) {
          closeLightbox();
        }
      });
    }

    if (document.readyState === "loading") {
      document.addEventListener("DOMContentLoaded", initGalleryLightbox);
    } else {
      initGalleryLightbox();
    }
  })();
</script>
